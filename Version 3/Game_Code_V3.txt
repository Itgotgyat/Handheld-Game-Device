int getCurrentSlotOwner() {

    return s03;
  
  if (r == 0 && c == 6) 
    return s06;
  if (r == 3 && c == 0) 
    return s30;
  
  if (r == 3 && c == 3) 
    return s33;
  
  if (r == 3 && c == 6) 
    return s36;
  if (r == 6 && c == 0) 
    return s60;
  
  if (r == 6 && c == 3) 
    return s63;
  if (r == 6 && c == 6) 
    return s66;

  
  return 0;
}

void Quit() {
  lcd.clearDisplay(0);
  led.clear();
  led.setCursor(0, 0);
  led.print("Game Quit");

  gaming = false;
  gaming2 = false;
  s00 = s03 = s06 = 0;
  s30 = s33 = s36 = 0;
  s60 = s63 = s66 = 0;
  chance = 0;
  page = 1;
  r = 0;
  c = 0;
  p1 = 0;
  p2 = 0;
  selectingDiff = false;
  length = 3;
  x[0] = 1;
  x[1] = 2;
  x[2] = 3;
  y[0] = 4;
  y[1] = 4;
  y[2] = 4;
  trial = 0;
  velX = 1;
  velY = 0;
  lcd.clearDisplay(0);
  if(selectingSetting == true){
  lcd.clearDisplay(0);
  led.clear();
  led.setCursor(0, 0);
  }
  selectingSetting = false;
  delay(1000);

  
}

void score(){

  led.clear();
  led.setCursor(0, 0);
  led.print("Player 1: ");
  led.print(p1);
  led.setCursor(0, 1);
  led.print("Player 2: ");
  led.print(p2);  
  gaming = false;
  s00 = s03 = s06 = 0;
  s30 = s33 = s36 = 0;
  s60 = s63 = s66 = 0;
  chance = 0;
  page = 1;

  c = 0;
  delay(5000);
  led.clear();
  led.setCursor(0, 0);
  led.print("Press Select to");
  led.setCursor(0, 1);
  led.print("play again");
  delay(500);
}

void select() {
  if (page == 1 && !selectingDiff) {
    if (!gaming) {
      lcd.clearDisplay(0);
      led.clear();
      led.setCursor(0, 0);
      led.print("Game Confirmed");
      gaming = true;
      totalTurns = 0;
    } 
    else if(gaming == true){
      if (getCurrentSlotOwner() == 0) {
        if (chance == 0) {
          setCurrentSlotOwner(1);
          chance = 1;
        } else if (chance == 1) {
          setCurrentSlotOwner(2);
          chance = 0;
      }
          lcd.setLed(0, r, c, true);
          lcd.setLed(0, r + 1, c, true);
          lcd.setLed(0, r, c + 1, true);
          lcd.setLed(0, r + 1, c + 1, true);
        totalTurns += 1;
  }
    }
  }

  if(page == 2 && !gaming && !selectingDiff){
    selectingDiff = true;
    page = 0;
    return;
  }
    if(selectingDiff){
      if(page == 0){
        speed = 500;
      }
      if(page == 1){
        speed = 350;
      }
      if(page == 2){
        speed = 250;
      }
      selectingDiff = false;
      gaming2 = true;
      drawSnake();
      return;
    }
    if(page == 3 && !gaming && !gaming2 && !selectingDiff){
        selectingSetting = true;
        return;
    }
}

void left() {
  if(!gaming && !gaming2){
    return;
  }
  if(page == 1) {
    clear();
    if(c > 0) {
    c -= 3;
    }
  }
  if(gaming2){
  if(velX != 1) {
  velX = -1;
  velY = 0;
            }
}
}

void right() {
  if (!gaming && !gaming2)
    return;
  if(gaming){
    clear();
  if (c < 6) {
      c += 3;
  }
  }
  if(gaming2){
    if(velX != -1) {
        velX = 1;
        velY = 0;
}
}
}

void up() {
  if (gaming){
    if(page == 1){
      clear();
    if (r > 0) {
    r -= 3;
}
}
}
  else if(gaming2){
    if(velY != 1){
    velY = -1;
    velX = 0;
}
  }
  else if(selectingSetting){
    brightness++;
    if(brightness > 10){
      brightness = 10;
    }
    lcd.setIntensity(0, brightness);
  }
}

void down() {

  if(selectingDiff){
    page++;
    if(page > 2){
      page = 0;
    }
  }
  else if(selectingSetting){
    brightness--;
    if(brightness < 0){
      brightness = 0;
    }
    lcd.setIntensity(0, brightness);
  }

  else if (gaming) {
    if (page == 1) {
      clear();
      if (r < 6) {
        r += 3;
  }
 }
  } else if (!gaming && !gaming2) {
    page++;
    if (page > 3) 
    page = 1;
  }
  else if(gaming2 && !selectingDiff){
    if(velY != -1){
    velY = 1;
    velX = 0;
}
 
  }
}

int foodX = 5;
int foodY = 5;
bool collision = false;

void snake() {
  if (millis() - lastMove > speed) {
    int tailX = x[0];
    int tailY = y[0];

    checkSnake();

    if (x[length - 1] == foodX && y[length - 1] == foodY) {
      foodEaten = true;
    } else {
      lcd.setLed(0, tailY, tailX, false);
      for (int col = 0; col < length - 1; col++) {
        x[col] = x[col + 1];
        y[col] = y[col + 1];
      }
    }
 if (foodEaten) {
  collision = true;
  //Gemini AI
 if (length <= 64) {
  length++; 
     x[length - 1] = x[length - 2];
      y[length - 1] = y[length - 2];
      } 
//-------------------------------------------------------------------------------------- 
      else {
        lcd.setLed(0, tailY, tailX, false);
    for (int col = 0; col < length - 1; col++) {
     x[col] = x[col + 1];
      y[col] = y[col + 1];
}
} 

  while(collision){
    collision = false;
  foodX = random(0, 8);
  foodY = random(0, 8);
  for(int i = 0; i<length; i++){
    if(x[i] == foodX && y[i] == foodY){
      collision = true;
      break;
    }

  }
  }
  lcd.setLed(0, foodY, foodX, true);
  foodEaten = false;
 }
  x[length - 1] += velX;
  y[length - 1] += velY;
    if (x[length - 1] > 7 || x[length - 1] < 0 || y[length - 1] > 7 || y[length - 1] < 0) {
      lcd.clearDisplay(0);
      Quit();
    }
    
  lcd.setLed(0, y[length - 1], x[length - 1], true);
  if (trial == 0) {
   lcd.setLed(0, foodY, foodX, true);
   }
    trial++;
    lastMove = millis();
  }
}

void checkSnake(){
  for(int i = 0; i<length - 1; i++){
    if(x[length - 1] == x[i] && y[length - 1] == y[i]){
      Quit();
      return;
    }
  }
}

void drawSnake(){
for(int col = 0; col<length; col++){
  lcd.setLed(0, y[col], x[col], true);
}
}


void setting(){
  if(first == 0){
  led.clear();
  }
  led.setCursor(0, 0);
  led.print("LED Brightness:");
  led.setCursor(0, 1);
  led.print(brightness);
  if(brightness < 10){
    led.setCursor(1, 1);
    led.print(" ");
  }
  lcd.setLed(0, 3, 3, true);
  lcd.setLed(0, 4, 3, true);
  lcd.setLed(0, 3, 4, true);
  lcd.setLed(0, 4, 4, true);
  first++;
}





void setup() {
  Serial.begin(9600);
  button.attachClick(select);
  bUp.attachClick(up);
  bRight.attachClick(right);
  bLeft.attachClick(left);
  bDown.attachClick(down);

  button.setPressMs(4000);
  button.attachLongPressStart(Quit);

  lcd.shutdown(0, false);
  lcd.setIntensity(0, 10);
  lcd.clearDisplay(0);

  led.begin(16, 2);
}

void loop() {
  button.tick();
  bRight.tick();
  bUp.tick();
  bLeft.tick();
  bDown.tick();


  if (page != lastPage && !gaming && !gaming2 && !selectingDiff) {
    led.clear();
    lcd.clearDisplay(0);
    if (page == 0) {
      led.setCursor(0, 0);
      led.print("Welcome");
      led.setCursor(0, 1);
      led.print("Start");
    }
    if (page == 1) {
      led.setCursor(0, 0);
      led.print("1: Tic Tac Toe");
      led.setCursor(0, 1);
      led.print("Next page");
      
    }
    if (page == 2) {
      led.setCursor(0, 0);
      led.print("2: Snake");
      led.setCursor(0, 1);
      led.print("Next page");
    }
    if (page == 3) {
      led.setCursor(0, 0);
      led.print("3: Settings");
      led.setCursor(0, 1);
      led.print("Next page");
    }
    lastPage = page;
  }
    if(selectingDiff){
      if (page != lastPage) {
        led.clear();
        led.setCursor(0, 0);
        if(page == 0) 
        led.print("Easy");
        else if(page == 1) 
        led.print("Medium");
        else if(page == 2) 
        led.print("Hard");
        
        led.setCursor(0, 1);
        led.print("Next Difficulty");
        
        lastPage = page;
      }
    }

    

  if (gaming && page == 1 && !selectingDiff) {
    ticTacToe();
  }
    if (gaming2 && !selectingDiff) {
    snake();
  }
  if(selectingSetting && !gaming && !gaming2 && !selectingDiff){
    setting();
  }
}
